 * 東京 7 天行程 Web App - 後端 API (動態掃描所有日期欄位，完美整合體能與步數預警)
 */ 

function doGet(e) {
  try {
    const ss = SpreadsheetApp.getActiveSpreadsheet();
    const scheduleSheet = ss.getSheetByName('Main_TravelPlan');
    const dbSheet = ss.getSheetByName('Database');
    const transSheet = ss.getSheetByName('Transport'); 

    if (!scheduleSheet || !dbSheet) {
      throw new Error("找不到 Main_TravelPlan 或 Database 工作表！");
    } 

    const scheduleData = parseBlockSchedule(scheduleSheet);
    const dbMapped = parseDatabase(dbSheet);
    const transMap = transSheet ? parseTransportSheet(transSheet) : {}; 

    const result = {
      status: 'success',
      updatedAt: new Date().toLocaleString("zh-TW", { timeZone: "Asia/Tokyo" }),
      schedule: scheduleData,
      restaurants: dbMapped.restaurants,
      attractions: dbMapped.attractions,
      transport: transMap
    }; 

    return ContentService.createTextOutput(JSON.stringify(result))
      .setMimeType(ContentService.MimeType.JSON); 

  } catch (err) {
    return ContentService.createTextOutput(JSON.stringify({
      status: 'error',
      message: err.toString()
    })).setMimeType(ContentService.MimeType.JSON);
  }
} 

/**
 * 1. 核心體能計算子函數：改用動態標題與安全對應抓取 Transport 數據
 */
function calculateFitnessForItems(items) {
  if (!items || items.length === 0) {
    return {
      totalSteps: 0,
      totalWalkMin: 0,
      totalDistM: 0,
      status: "🟢 輕鬆漫遊",
      summaryText: "🟢 輕鬆漫遊 | 預估步數: 0 步 | 總步行: 0 分鐘 (0.0 km)"
    };
  }

  const ss = SpreadsheetApp.getActiveSpreadsheet();
  const dbSheet = ss.getSheetByName("Database");
  const transportSheet = ss.getSheetByName("Transport");

  // 建立 Database 查找對照表
  const dbMap = {};
  if (dbSheet) {
    const dbData = dbSheet.getDataRange().getValues();
    for (let i = 1; i < dbData.length; i++) {
      const name = String(dbData[i][1]).trim();
      const indoorLoad = dbData[i][40] !== undefined ? String(dbData[i][40]).trim().toLowerCase() : 'light';
      dbMap[name] = indoorLoad;
    }
  }

  // 建立 Transport 查找對照表（使用動態欄位標題尋找，避免硬編碼出錯）
  const transMap = {};
  if (transportSheet) {
    const tData = transportSheet.getDataRange().getValues();
    if (tData.length > 1) {
      const headers = tData[0].map(h => String(h).trim());
      
      // 定義欄位索引：優先尋找對應名稱，若找不到則採用你指定的預設對應 (T=19, P=15, Q=16, R=17)
      let colRouteKey = headers.indexOf('Route_Key');
      if (colRouteKey === -1) colRouteKey = 19; // T 欄

      let colDistM = headers.indexOf('walk_dist_m');
      if (colDistM === -1) colDistM = 15; // P 欄

      let colTimeMin = headers.indexOf('walk_time_min');
      if (colTimeMin === -1) colTimeMin = 16; // Q 欄

      let colDiffLoad = headers.indexOf('difficulty_load');
      if (colDiffLoad === -1) colDiffLoad = 17; // R 欄

      for (let i = 1; i < tData.length; i++) {
        const row = tData[i];
        const routeKey = String(row[colRouteKey] || '').trim();
        if (!routeKey) continue;

        const distM = Number(row[colDistM]) || 0;
        const timeMin = Number(row[colTimeMin]) || 0;
        const diffLoad = Number(row[colDiffLoad]) || 0;

        transMap[routeKey] = { distM, timeMin, diffLoad };
        
        // 同時支援雙向或常見的符號替換
        transMap[routeKey.replace(/➔/g, '→')] = { distM, timeMin, diffLoad };
        transMap[routeKey.replace(/→/g, '➔')] = { distM, timeMin, diffLoad };
      }
    }
  }

  let totalDistM = 0;
  let totalWalkMin = 0;
  let totalSteps = 0;

  for (let i = 0; i < items.length; i++) {
    const item = items[i];
    const name = item.name;
    const isTransport = name.includes('→') || name.includes('➔');

    if (isTransport) {
      const cleanRoute = name.replace(/→/g, '➔').trim();
      const tInfo = transMap[cleanRoute] || transMap[cleanRoute.replace(/➔/g, '→')] || { distM: 0, timeMin: 5, diffLoad: 0 };
      
      let legDist = tInfo.distM;
      let legTime = tInfo.timeMin + tInfo.diffLoad; // 將 difficulty_load 加進步行時間
      
      if (legDist === 0 && legTime > 0) {
        legDist = legTime * 75; // 若沒有填寫距離，以每分鐘 75 米估算
      }

      totalDistM += legDist;
      totalWalkMin += legTime;
      totalSteps += (legDist / 0.7); // 以每步 0.7 米計算

    } else {
      const durationMins = Number(item.duration) || 60;
      const indoorLoad = dbMap[name] || 'light';
      
      let ratePerH = 150;
      if (indoorLoad === 'moderate') ratePerH = 500;
      if (indoorLoad === 'heavy') ratePerH = 1200;

      const indoorDist = (durationMins / 60) * ratePerH;
      const indoorTime = indoorDist / 75;

      totalDistM += indoorDist;
      totalWalkMin += indoorTime;
      totalSteps += (indoorDist / 0.7);
    }
  }

  totalDistM = Math.round(totalDistM);
  totalWalkMin = Math.round(totalWalkMin);
  totalSteps = Math.round(totalSteps);

  let status = "🟢 輕鬆漫遊";
  if (totalSteps >= 12000 && totalSteps < 15000) {
    status = "🟡 步行適中 (注意長輩體力)";
  } else if (totalSteps >= 15000) {
    status = "🔴 體能超載警示 (>1.5萬步，建議加插休息)";
  }

  const summaryText = `${status} | 預估步數: ${totalSteps.toLocaleString()} 步 | 總步行: ${totalWalkMin} 分鐘 (${(totalDistM/1000).toFixed(1)} km)`;

  return {
    totalSteps: totalSteps,
    totalWalkMin: totalWalkMin,
    totalDistM: totalDistM,
    status: status,
    summaryText: summaryText
  };
}
/**
 * 2. 行程結構解析函數 (使用動態逐欄掃描，完整對應所有日期區塊)
 */
function parseBlockSchedule(sheet) {
  const data = sheet.getDataRange().getValues();
  const displayData = sheet.getDataRange().getDisplayValues();
  
  if (data.length < 2) return { days: [], timeline: {} }; 

  const headerRow = data[0];
  const daysData = {};
  const daysList = []; 

  // 💡 改為逐欄掃描，確保不會漏掉任何一天的區塊
  for (let c = 0; c < headerRow.length; c++) {
    let cellVal = headerRow[c];
    if (cellVal instanceof Date) {
      cellVal = Utilities.formatDate(cellVal, "Asia/Tokyo", "yyyy-MM-dd");
    }
    const headerStr = String(cellVal).trim(); 

    if (/^\d{4}-\d{2}-\d{2}$/.test(headerStr)) {
      const dayKey = headerStr;
      daysList.push(dayKey); 

      const timeCol = c;
      const nameCol = c + 1;
      const durationCol = c + 2;
      const departureCol = c + 3; 

      // 動態定位 "Time" 或 "Name" 標題列
      let startRow = 2; 
      for (let r = 0; r < data.length; r++) {
        const val = data[r][nameCol] !== undefined ? String(data[r][nameCol]).trim() : '';
        if (val === 'Name' || val === 'Time') {
          startRow = r + 1; 
          break;
        }
      }

      const items = []; 
      for (let r = startRow; r < data.length; r++) {
        if (!data[r] || nameCol >= data[r].length) continue;
        
        const nameRaw = data[r][nameCol] !== undefined ? String(data[r][nameCol]).trim() : ''; 
        // 🛡️ 略過空白、NaN、標題，同時過濾掉體能預警摘要文字行以免被誤認為景點
        if (!nameRaw || nameRaw === 'NaN' || nameRaw === 'Name' || nameRaw === 'Time' || nameRaw.includes('🟢') || nameRaw.includes('🟡') || nameRaw.includes('🔴')) {
          continue;
        }

        let spotName = nameRaw;
        let notice = '';
        if (nameRaw.includes('|')) {
          const parts = nameRaw.split('|');
          spotName = parts[0].trim();
          notice = parts.slice(1).join('|').trim();
        } 

        let arrivalTime = displayData[r][timeCol] ? String(displayData[r][timeCol]).trim() : '';
        let departureTime = displayData[r][departureCol] ? String(displayData[r][departureCol]).trim() : '';

        arrivalTime = cleanTimeString(arrivalTime);
        departureTime = cleanTimeString(departureTime);

        const durationVal = data[r][durationCol] !== '' && !isNaN(data[r][durationCol]) ? Number(data[r][durationCol]) : 0;

        items.push({
          time: arrivalTime,
          name: spotName,
          duration: durationVal,
          departure: departureTime,
          notice: notice
        });
      } 

      // 計算該日體能數據並封裝
      const fitnessInfo = calculateFitnessForItems(items);

      daysData[dayKey] = {
        items: items,
        fitness: fitnessInfo
      };
    }
  } 

  return {
    days: daysList,
    timeline: daysData
  };
}

/**
 * 3. 試算表自訂函數
 */
function CALCULATE_DAY_FITNESS(dayKey) {
  if (dayKey instanceof Date) {
    dayKey = Utilities.formatDate(dayKey, "Asia/Tokyo", "yyyy-MM-dd");
  } else {
    dayKey = String(dayKey).trim();
  }
  const ss = SpreadsheetApp.getActiveSpreadsheet();
  const sheet = ss.getSheetByName("Main_TravelPlan");
  if (!sheet) return "🟢 輕鬆漫遊 | 預估步數: 0 步 | 總步行: 0 分鐘 (0.0 km)";
  
  const parsed = parseBlockSchedule(sheet);
  const dayObj = parsed.timeline[dayKey];
  if (!dayObj) return "🟢 輕鬆漫遊 | 預估步數: 0 步 | 總步行: 0 分鐘 (0.0 km)";
  
  return dayObj.fitness;
}

function GET_DAY_FITNESS_SUMMARY(dayKey) {
  const res = CALCULATE_DAY_FITNESS(dayKey);
  if (typeof res === 'string') return res;
  return res.summaryText;
}

/**
 * 【輔助函數】清洗時間字串
 */
function cleanTimeString(str) {
  if (!str) return '';
  if (str.includes('1899') || str.includes('T')) {
    const match = str.match(/(\d{2}:\d{2})/);
    if (match) return match[1];
  }
  return str;
}

function formatCellTimeCumulative(val) {
  if (val === null || val === undefined || val === '') return '';

  if (val instanceof Date) {
    const hours = String(val.getUTCHours()).padStart(2, '0');
    const minutes = String(val.getUTCMinutes()).padStart(2, '0');
    return `${hours}:${minutes}`;
  }

  if (typeof val === 'number') {
    const totalMinutes = Math.round(val * 24 * 60);
    const hours = Math.floor(totalMinutes / 60); 
    const minutes = totalMinutes % 60;
    return `${String(hours).padStart(2, '0')}:${String(minutes).padStart(2, '0')}`;
  }

  return String(val).trim();
}

// 解析 Database
function parseDatabase(sheet) {
  const data = sheet.getDataRange().getValues();
  if (data.length < 2) return { restaurants: {}, attractions: {} }; 

  const rawHeaders = data[0];
  const headers = [];
  let lastHeader = ''; 

  const dayMap = {
    '日': '日', '天': '日', '週日': '日', '星期日': '日',
    '一': '一', '週一': '一', '星期一': '一',
    '二': '二', '週二': '二', '星期二': '二',
    '三': '三', '週三': '三', '星期三': '三',
    '四': '四', '週四': '四', '星期四': '四',
    '五': '五', '週五': '五', '星期五': '五',
    '六': '六', '週六': '六', '星期六': '六'
  }; 

  for (let j = 0; j < rawHeaders.length; j++) {
    let rawH = String(rawHeaders[j]).trim();
    let h = rawH; 

    if (rawH !== '') {
      const dayMatch = rawH.match(/^(?:星期|週)?([日一二三四五六天])/);
      if (dayMatch && dayMap[dayMatch[1]]) {
        h = dayMap[dayMatch[1]];
      }
      lastHeader = h;
      headers.push(h);
    } else {
      headers.push(lastHeader);
    }
  } 

  const restaurants = {};
  const attractions = {}; 

  for (let i = 1; i < data.length; i++) {
    const rowObj = {};
    for (let j = 0; j < headers.length; j++) {
      const key = headers[j];
      if (!key) continue; 

      let val = data[i][j] !== undefined ? String(data[i][j]).trim() : '';
      if (data[i][j] instanceof Date) {
        val = Utilities.formatDate(data[i][j], "Asia/Tokyo", "HH:mm");
      } 

      if (val !== '') {
        if (rowObj[key]) {
          rowObj[key] += ' ' + val;
        } else {
          rowObj[key] = val;
        }
      } else if (!rowObj.hasOwnProperty(key)) {
        rowObj[key] = '';
      }
    } 

    for (let k in rowObj) {
      if (typeof rowObj[k] === 'string') {
        rowObj[k] = rowObj[k].replace(/\s+/g, ' ').trim();
      }
    } 

    const name = rowObj.Name;
    if (!name) continue; 

    const type = rowObj.Type || '';
    if (type === '餐廳' || type === '面包/甜點') {
      restaurants[name] = rowObj;
    } else {
      attractions[name] = rowObj;
    }
  } 

  return { restaurants: restaurants, attractions: attractions };
} 

// 解析 Transport
function parseTransportSheet(sheet) {
  const data = sheet.getDataRange().getValues();
  if (data.length < 2) return {}; 

  const headers = data[0].map(h => String(h).trim());
  const map = {}; 

  for (let i = 1; i < data.length; i++) {
    const rowObj = {};
    for (let j = 0; j < headers.length; j++) {
      rowObj[headers[j]] = data[i][j] !== undefined ? String(data[i][j]).trim() : '';
    } 

    const fromVal = rowObj.From || '';
    const toVal = rowObj.To || ''; 

    if (fromVal && toVal) {
      const routeKey = `${fromVal}➔${toVal}`;
      map[routeKey] = rowObj;
      map[fromVal] = rowObj;
    }
  }
  return map;
}
